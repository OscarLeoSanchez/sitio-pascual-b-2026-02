Inferencia estadística

ANOVA de una vía

Cómo saber si tres o más grupos tienen la misma media, o si alguno se separa.

from scipy import stats

alfa = 0.05
sin_tutoria = [2.6, 3.0, 3.2, 2.8, 3.4, 3.0]
grupal = [3.4, 3.8, 3.6, 4.0, 3.2, 3.6]
individual = [3.8, 4.2, 4.0, 4.4, 3.6, 4.0]

F, p = stats.f_oneway(sin_tutoria, grupal, individual)
print(f"F = {F:.2f}  p = {p:.5f}")
print("Se rechaza H0" if p < alfa else "No se rechaza H0")
Salida
F = 19.00  p = 0.00008
Se rechaza H0
from scipy import stats

nombres = ["sin tutoria", "grupal", "individual"]
sin_tutoria = [2.6, 3.0, 3.2, 2.8, 3.4, 3.0]
grupal = [3.4, 3.8, 3.6, 4.0, 3.2, 3.6]
individual = [3.8, 4.2, 4.0, 4.4, 3.6, 4.0]

r = stats.tukey_hsd(sin_tutoria, grupal, individual)
for i, j in [(0, 1), (0, 2), (1, 2)]:
    dif = r.statistic[j][i]
    p = r.pvalue[i][j]
    print(f"{nombres[i]} y {nombres[j]}: diferencia = {dif:.2f}  p = {p:.3f}")
Salida
sin tutoria y grupal: diferencia = 0.60  p = 0.006
sin tutoria y individual: diferencia = 1.00  p = 0.000
grupal y individual: diferencia = 0.40  p = 0.066
