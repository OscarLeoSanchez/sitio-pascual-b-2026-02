class Libro
{
    public string Titulo { get; }
    public string Autor { get; }
    public int Copias { get; private set; }

    public Libro(string titulo, string autor, int copias)
    {
        Titulo = titulo;
        Autor = autor;
        Copias = copias < 0 ? 0 : copias;
    }

    public bool Prestar()
    {
        if (Copias == 0) return false;
        Copias--;
        return true;
    }

    public void Devolver()
    {
        Copias++;
    }
}
class Libro:
    def __init__(self, titulo, autor, copias):
        self.titulo = titulo
        self.autor = autor
        self._copias = max(0, copias)

    @property
    def copias(self):
        return self._copias

    def prestar(self):
        if self._copias == 0:
            return False
        self._copias -= 1
        return True

    def devolver(self):
        self._copias += 1
public class Libro {
    private final String titulo;
    private final String autor;
    private int copias;

    public Libro(String titulo, String autor, int copias) {
        this.titulo = titulo;
        this.autor = autor;
        this.copias = copias < 0 ? 0 : copias;
    }

    public String getTitulo() {
        return titulo;
    }

    public int getCopias() {
        return copias;
    }

    public boolean prestar() {
        if (copias == 0) return false;
        copias--;
        return true;
    }

    public void devolver() {
        copias++;
    }
}

class Usuario

    public string Nombre { get; }
    public int Prestamos { get; private set; }

    public Usuario(string nombre)
    {

    public bool PedirPrestado(Libro libro)

        Prestamos++;
        return true;
    }

        return nombre;
    }

    public int getPrestamos() {
        return prestamos;
    }

    public boolean pedirPrestado(Libro libro) {
        if (prestamos >= 2) return false;
        if (!libro.prestar()) return false;
        prestamos++;
        return true;
    }
}
